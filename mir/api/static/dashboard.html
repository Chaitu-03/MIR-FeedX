    Sample Queries
    

      
Row counts:

      SELECT 'posts' AS tbl, COUNT(*) FROM posts UNION ALL SELECT 'accounts', COUNT(*) FROM accounts UNION ALL SELECT 'tags', COUNT(*) FROM tags UNION ALL SELECT 'communities', COUNT(*) FROM communities;
      
Post breakdown:

      SELECT COUNT(*) FILTER (WHERE nsfw=false) AS sfw, COUNT(*) FILTER (WHERE nsfw=true) AS nsfw, COUNT(*) FILTER (WHERE image_embedding IS NOT NULL) AS with_img, COUNT(*) FILTER (WHERE body_clean IS NOT NULL) AS with_text FROM posts;
      
Latest posts:

      SELECT p.id, a.blog_name, LEFT(p.body_clean, 80) AS body, p.note_count, p.nsfw, p.published_at FROM posts p JOIN accounts a ON a.id = p.account_id ORDER BY p.created_at DESC LIMIT 20;
      
Top tags:

      SELECT name, usage_count FROM tags ORDER BY usage_count DESC LIMIT 20;
      
Crawl progress:

      SELECT blog_name, status, last_crawled_at, fail_count, last_timestamp FROM crawl_state ORDER BY last_crawled_at DESC NULLS LAST;
      
Blogs with most posts:

      SELECT a.blog_name, COUNT(p.id) AS post_count FROM accounts a LEFT JOIN posts p ON p.account_id = a.id GROUP BY a.blog_name ORDER BY post_count DESC LIMIT 20;
